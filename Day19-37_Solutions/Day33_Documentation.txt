public class Day33_LinkedListCycle {

    static class ListNode {
        int val;
        ListNode next;

        ListNode(int val) {
            this.val = val;
            this.next = null;
        }
    }

    public boolean hasCycle(ListNode head) {
        if (head == null || head.next == null) {
            return false;
        }

        ListNode slow = head;
        ListNode fast = head;

        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;

            if (slow == fast) {
                return true;
            }
        }

        return false;
    }

    public ListNode detectCycle(ListNode head) {
        if (head == null || head.next == null) {
            return null;
        }

        ListNode slow = head;
        ListNode fast = head;

        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;

            if (slow == fast) {
                break;
            }
        }

        if (fast == null || fast.next == null) {
            return null;
        }

        slow = head;
        while (slow != fast) {
            slow = slow.next;
            fast = fast.next;
        }

        return slow;
    }

    public static ListNode createTestList() {
        ListNode node1 = new ListNode(3);
        ListNode node2 = new ListNode(2);
        ListNode node3 = new ListNode(0);
        ListNode node4 = new ListNode(-4);

        node1.next = node2;
        node2.next = node3;
        node3.next = node4;
        node4.next = node2;

        return node1;
    }

    public static void main(String[] args) {
        Day33_LinkedListCycle detector = new Day33_LinkedListCycle();

        ListNode listWithCycle = createTestList();
        System.out.println("List with cycle:");
        System.out.println("Has cycle: " + detector.hasCycle(listWithCycle));

        ListNode cycleStart = detector.detectCycle(listWithCycle);
        if (cycleStart != null) {
            System.out.println("Cycle starts at node with value: " + cycleStart.val);
        }

        ListNode listNoCycle = new ListNode(1);
        listNoCycle.next = new ListNode(2);
        listNoCycle.next.next = new ListNode(3);

        System.out.println("\nList without cycle:");
        System.out.println("Has cycle: " + detector.hasCycle(listNoCycle));

        ListNode noCycleStart = detector.detectCycle(listNoCycle);
        if (noCycleStart == null) {
            System.out.println("No cycle detected");
        }
    }
}
